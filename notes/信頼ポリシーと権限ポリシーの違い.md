# 信頼ポリシーと権限ポリシーの違い

- 信頼ポリシー
  - ロールに対して付与するもの
    - terraform上では
  - だれがこのロールを引き受けられるか を定義
  - 普通は `actions = ["sts:AssumeRole"]` のみを指定する
  - 1ロールにつき1信頼ポリシーのみ指定できる
- 権限ポリシー
  - 一般的な方。権限と言ってまず想像する方
  - 何ができるか を定義
  - `s3:PutObject` などいろいろ
**例.tf**

```
# =============================================
# Step 1: 権限ポリシーの中身を定義する
# =============================================

# 「何ができるか」の定義
data "aws_iam_policy_document" "s3_access" {
  statement {
    effect = "Allow"
    actions = [
      "s3:GetObject",
      "s3:PutObject",
    ]
    resources = ["arn:aws:s3:::my-bucket/*"]
  }
}

# =============================================
# Step 2: 信頼ポリシーの中身を定義する
# =============================================

# 「だれがこのロールを引き受けられるか」の定義
data "aws_iam_policy_document" "lambda_assume_role" {
  statement {
    effect  = "Allow"
    actions = ["sts:AssumeRole"] # ← 信頼ポリシーでは基本これだけ

    principals {
      type        = "Service"
      identifiers = ["lambda.amazonaws.com"] # ← Lambdaサービスが引き受けられる
    }
  }
}

# =============================================
# Step 3: ロールを作成し、信頼ポリシーを紐づける
# =============================================

resource "aws_iam_role" "lambda" {
  name = "my-lambda-role"

  # ↓ 信頼ポリシーをロールに紐づける（1ロールにつき1つだけ）
  assume_role_policy = data.aws_iam_policy_document.lambda_assume_role.json
}

# =============================================
# Step 4: 権限ポリシーをロールにアタッチする
# =============================================

resource "aws_iam_role_policy" "lambda_s3" {
  name = "lambda-s3-access"
  role = aws_iam_role.lambda.id

  # ↓ 権限ポリシーをロールに紐づける（複数アタッチ可）
  policy = data.aws_iam_policy_document.s3_access.json
}

```
