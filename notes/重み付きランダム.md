# 重み付きランダム

```js
// distributionTextで指定された分布に基づいて、0からduration - 1までのランダムな値を返す関数。
// たとえば下記のように呼び出すと、0〜99の全ての整数のうち、0寄りの値が選ばれやすいような分布でランダムな値を返す。
// (0〜24が4/10, 25〜49が3/10, 50〜74が2/10, 75〜99が1/10の確率で選ばれるイメージ)
// weightedRandom(`
//   ####
//   ###
//   ##
//   #
// `, 100)
export const weightedRandom = (distributionText, duration) => {
  // weights: [4, 3, 2, 1] のような、'#'の個数が入った配列
  const weights = distributionText
    .trim()
    .split('\n')
    .map((line) => line.trim().length)

  // weightIndexes: [0, 0, 0, 0, 1, 1, 1, 2, 2, 3] のような、'#'の個数個のインデックスが入った配列
  const weightIndexes = weights.flatMap((length, index) => Array(length).fill(index))

  const selectedWeightIndex = weightIndexes[rand(weightIndexes.length)]

  const durationUnit = Math.floor(duration / weights.length)

  // selectedWeightIndexに2が選ばれたとき、
  // randomRangeMin: 25 * 2 = 50
  // randomRangeMax: 25 * (2 + 1) = 75
  // となり、50〜74の範囲でランダムな値を返す
  const randomRangeMin = durationUnit * selectedWeightIndex
  const randomRangeMax = durationUnit * (selectedWeightIndex + 1)
  return randomRangeMin + rand(randomRangeMax - randomRangeMin)
}


```
